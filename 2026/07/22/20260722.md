function process_char(c) {
  while (1) {
    switch(state) {
      case "START_RECORD":
        if (c == "EOL") break
        if (is_crlf(c)) {
          save_field()
          state = "EAT_CRLF"
          break
        } else {
          state = "START_FIELD"
          continue
        }
      case "START_FIELD":
        if (c == "EOL" || is_crlf(c)) {
          save_field()
          if (c == "EOL") state = "START_RECORD"
          else state = "EAT_CRLF"
        } else if (c == ",") {
          save_field()
        } else if (c == "\"") {
          state = "IN_QUOTED_FIELD"
        } else {
          add_char(c)
          state = "IN_FIELD"
        }
        break
      case "IN_FIELD":
        if (c == "EOL" || is_crlf(c)) {
          save_field()
          if (c == "EOL") state = "START_RECORD"
          else state = "EAT_CRLF"
        } else if (c == ",") {
          save_field()
          state = "START_FIELD"
        } else {
          add_char(c)
        }
        break
      case "IN_QUOTED_FIELD":
        if (c == "EOL") {
          add_char("\n")
        } else if (c == "\"") {
          state = "QUOTE_IN_QUOTED_FIELD"
        } else {
          add_char(c)
        }
        break
      case "QUOTE_IN_QUOTED_FIELD":
        if (c == "EOL" || is_crlf(c)) {
          save_field()
          if (c == "EOL") state = "START_RECORD"
          else state = "EAT_CRLF"
        } else if (c == ",") {
          save_field()
          state = "START_FIELD"
        } else if (c == "\"") {
          add_char(c)
          state = "IN_QUOTED_FIELD"
        } else {
          error(UNEXPECTED_CHAR_AFTER_END_QUOTE)
        }
        break
      case "EAT_CRLF":
        if (is_crlf(c)) break
        else if (c == "EOL") {
          state = "START_RECORD"
          break
        } else {
          error(UNEXPECTED_CHAR_AFTER_CRLF)
        }
    }
    break
  }
}
